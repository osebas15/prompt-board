## Implementation Plan

### Phase 1: Analytics Foundation (Day 1-2)
1. **Analytics Database Schema**
   ```sql
   -- Usage tracking table
   CREATE TABLE usage_analytics (
     id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
     user_id UUID REFERENCES auth.users(id),
     prompt_id UUID REFERENCES prompts(id),
     action_type TEXT NOT NULL, -- 'view', 'create', 'edit', 'share', 'comment'
     metadata JSONB,
     session_id TEXT,
     created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
   );
   
   -- Aggregated metrics table for performance
   CREATE TABLE daily_metrics (
     date DATE NOT NULL,
     organization_id UUID REFERENCES organizations(id),
     metric_type TEXT NOT NULL,
     metric_value INTEGER NOT NULL,
     metadata JSONB,
     PRIMARY KEY (date, organization_id, metric_type)
   );
   
   -- Performance monitoring
   CREATE TABLE performance_metrics (
     id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
     metric_name TEXT NOT NULL,
     value NUMERIC NOT NULL,
     tags JSONB,
     recorded_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
   );
   ```

2. **Analytics Collection System**
   ```typescript
   // useAnalytics.ts
   export const useAnalytics = () => {
     const trackEvent = useCallback(async (event: AnalyticsEvent) => {
       try {
         await supabase.from('usage_analytics').insert({
           user_id: currentUser?.id,
           prompt_id: event.promptId,
           action_type: event.action,
           metadata: event.metadata,
           session_id: sessionId
         });
       } catch (error) {
         // Fail silently for analytics
         console.warn('Analytics tracking failed:', error);
       }
     }, [currentUser, sessionId]);
     
     return { trackEvent };
   };
   
   // Analytics event types
   interface AnalyticsEvent {
     action: 'view' | 'create' | 'edit' | 'share' | 'comment' | 'search';
     promptId?: string;
     metadata?: Record<string, any>;
   }
   ```
